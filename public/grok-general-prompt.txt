    s2rgb = nodes.new("ShaderNodeShaderToRGB")
    links.new(bsdf.outputs["BSDF"], s2rgb.inputs["Shader"])
    links.new(s2rgb.outputs["Color"], output.inputs["Surface"])  ← DIRECT to Output

    ls.select_external_contour = True

  1. bpy.ops.import_scene.gltf(filepath=PATH)

       for corner in mesh.bound_box:

  4. Scale: sf = target_height / native_height
     obj.scale = (obj.scale.x * sf, obj.scale.y * sf, obj.scale.z * sf)
  5. bpy.context.view_layer.update()  ← REQUIRED after scale
  6. Re-measure, then ground: obj.location.z -= z_min

  ── PHASE 2: HERO CHARACTER ────────────────────────────────────────────────
  3. Character probe loop (try models in priority order):
       HERO_SEARCH_ORDER = [
           "9263ca597dea49bb8bb249f21c7bdc4d.glb",   # ★ PROVEN anime girl, sf=0.267
           "b2359160a4f54e76b5ae427a55d9594d.glb",   # ✗ NOT humanoid — all env objects
       ]
       DOWNLOADS = r"C:\Users\Aiden\Downloads"
       for glb in HERO_SEARCH_ORDER:
           path = os.path.join(DOWNLOADS, glb)
           if not os.path.exists(path): continue
           size_mb = os.path.getsize(path) / 1e6
           print(f"[HERO_TRY] {glb} ({size_mb:.1f}MB)")
           bpy.ops.import_scene.gltf(filepath=path)
           bpy.context.view_layer.update()
           # Count meshes and armatures
           meshes = [o for o in bpy.data.objects if o.type=='MESH']
           arms   = [o for o in bpy.data.objects if o.type=='ARMATURE']
           # Remove blockers (objects taller than 3m world-space)
           removed = 0
           for obj in meshes[:]:
               bb = [obj.matrix_world @ Vector(c) for c in obj.bound_box]
               h = max(v.z for v in bb) - min(v.z for v in bb)
               if h > 3.0:
                   bpy.data.objects.remove(obj, do_unlink=True); removed += 1
           bpy.context.view_layer.update()
           meshes2 = [o for o in bpy.data.objects if o.type=='MESH']
           arms2   = [o for o in bpy.data.objects if o.type=='ARMATURE']
           print(f"[HERO_PROBE] meshes={len(meshes2)} arms={len(arms2)} "
                 f"blockers_removed={removed}")
           if len(meshes2) == 0 or len(arms2) == 0:
               print(f"[HERO_SKIP] Not a character, trying next")
               # Clean up and try next
               for o in list(bpy.data.objects): bpy.data.objects.remove(o,do_unlink=True)
               for c in [bpy.data.meshes,bpy.data.materials,bpy.data.armatures]:
                   for x in list(c): c.remove(x)
               continue
           # Measure + scale to TARGET_HEIGHT (1.6m)
           all_objs = [o for o in bpy.data.objects if o.type in ('MESH','ARMATURE')]
           bb_all = []
           for o in all_objs:
               bb_all += [o.matrix_world @ Vector(c) for c in o.bound_box]
           z_min = min(v.z for v in bb_all); z_max = max(v.z for v in bb_all)
           native_h = z_max - z_min
           sf = 1.6 / native_h if native_h > 0.001 else 1.0
           center = Vector((
               (max(v.x for v in bb_all)+min(v.x for v in bb_all))/2,
               (max(v.y for v in bb_all)+min(v.y for v in bb_all))/2,
               z_min))
           for o in all_objs:
               o.scale *= sf
               o.location -= center * sf
               o.location.z -= z_min * sf
           bpy.context.view_layer.update()
           print(f"[HERO_OK] Using {glb}, sf={sf:.3f}, center={center}")
           HERO_GLB = glb
           break

       for mat in bpy.data.materials:

               out = next((n for n in nodes if n.type=='OUTPUT_MATERIAL'),None)

               if out and bsdf:

§23A  FRACTURE: MASTER CHROME CDP PATTERN (proven across all web tasks)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
EVERY web task follows this exact sequence:

  STEP 1 — Launch Chrome (ONCE per session, NOT for subsequent navigation):
    chain=run:start chrome https://INITIAL_URL
    ← This auto-enables --remote-debugging-port=9222 for CDP
    ← TRAP: NEVER use "start chrome" to navigate again — it spawns a new
      process and breaks CDP. For all subsequent pages use nav: instead.

  STEP 2 — Wait for page load:
    chain=....|wait:6000   ← 6s for normal pages, 8s for Google/SPAs

  STEP 3 — Read the page (MANDATORY before clicking anything):
    chain=....|snapshot
    ← Returns: url, title, bodyText, element map with real CSS selectors
    ← RULE: Never guess selectors. Always snapshot first, use what it returns.

  STEP 4 — Navigate to next page (use nav: NOT start chrome):
    chain=nav:https://NEW_URL|wait:6000|snapshot

  STEP 5 — Click element by selector:
    chain=click:button[aria-label='Play']
    ← TRAP: Third-party sites change DOM. If selector fails, re-snapshot.

  STEP 6 — Run JavaScript in page:
    chain=eval:document.querySelector('button').click()
    ← For complex interactions: find elements by text content when selectors fail:
      eval:([...document.querySelectorAll('button')].find(b=>b.textContent.trim()==='Off')||{click:()=>{}}).click()

  STEP 7 — Type into input:
    chain=type_text:input[name='search']>>>YOUR TEXT

  STEP 8 — CDP mouse (canvas/drawing apps — page coords, no DPI scaling):
    cdp_drag:X1,Y1,X2,Y2        ← drag on canvas
    cdp_scroll:X,Y,DELTA_Y      ← scroll (neg=up, pos=down)
    cdp_double_click:SELECTOR   ← double-click by selector or coords

  STEP 9 — Verify on stream (non-negotiable):
    GET {T}/api/desktop-stream   ← YOUR EYES — check this, not API JSON
    chain=screenshot:Chrome      ← or single frame snapshot
    GET {T}/api/desktop-frame?source=cdp&nogrid=1&t=TIMESTAMP

  POWER PATTERN — combine all steps in ONE grok-do call:
    ?chain=run:start chrome https://URL|wait:6000|snapshot|eval:SCRIPT|screenshot:Chrome

  CDP CLICK ACCURACY (3840×2160 desktop, DPI 2.5×):
    physicalX = CSS_X × 2.5
    physicalY = CSS_Y × 2.5 + 150   ← Chrome UI offset (tabs + address bar)
    Calibrated 135/135 clicks — 100% accuracy with this formula.

§23G  KEY LESSONS FROM WEB SESSIONS (sourced from lessons-learned.md)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  ✗ Hardcoded single selector on third-party UI → Use 6-8 fallback selector variants
  ✗ start chrome to re-navigate → CDP breaks. Use nav: chain step after first launch.
  ✗ CDP click() on SoundCloud play → Focus loss, track pauses. Use hw.exe click.
  ✗ Multiple agents sending chunks in parallel → Race condition, file corruption.
      Always use ONE agent (single-executor rule). Never parallelize sequential writes.
  ✗ grok-write on Windows files with \r\n endings → silent 0 replacements.
      Use grok-create (full overwrite) for any file on the Windows desktop project.
  ✗ content > 1000 chars in a single URL → URL truncated silently.
      Use grok-create-chunk with ~800-char chunks. Final chunk triggers auto-verify.
  ✗ browse_page summarizer misreads JSON as file content → Don't use char counts.
      Verify with content markers: "starts with import", "has key string", not char count.
  ✗ React SPA state not updated immediately after click → wait:2000 before verifying.
